cmake_minimum_required(VERSION 3.28)
project(ozenproto LANGUAGES C)

set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
# CMake 3.28 spells C23 as -std=c2x for clang 17; the plan pins -std=c23 for clang (D13).
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
  set(CMAKE_C23_STANDARD_COMPILE_OPTION "-std=c23")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(OZP_SANITIZE "Build with AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(OZP_FUZZ "Build the libFuzzer target" OFF)

add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wstrict-prototypes -Werror)
if(OZP_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined -fno-sanitize-recover=all)
endif()

file(GLOB OZP_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.c)
add_library(ozenproto_lib STATIC ${OZP_SOURCES})
set_target_properties(ozenproto_lib PROPERTIES OUTPUT_NAME ozenproto)
target_include_directories(ozenproto_lib PUBLIC include PRIVATE src)

add_executable(ozenproto driver/main.c)
target_link_libraries(ozenproto PRIVATE ozenproto_lib)

enable_testing()
file(GLOB OZP_UNIT_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/unit/*.c)
add_executable(ozp_unit ${OZP_UNIT_SOURCES})
target_include_directories(ozp_unit PRIVATE src)
target_link_libraries(ozp_unit PRIVATE ozenproto_lib)
file(STRINGS ${CMAKE_CURRENT_SOURCE_DIR}/tests/unit/tests.def OZP_UNIT_LINES REGEX "^OZP_TEST\\(")
set(OZP_UNIT_TESTS "")
foreach(l IN LISTS OZP_UNIT_LINES)
  string(REGEX REPLACE "^OZP_TEST\\(([a-z0-9_]+)\\)$" "\\1" n "${l}")
  list(APPEND OZP_UNIT_TESTS "${n}")
endforeach()
foreach(t IN LISTS OZP_UNIT_TESTS)
  if(t STREQUAL "arena_poison_detects")
    if(OZP_SANITIZE)
      add_test(NAME ${t} COMMAND ozp_unit ${t})
      set_tests_properties(${t} PROPERTIES WILL_FAIL TRUE)
    endif()
  else()
    add_test(NAME ${t} COMMAND ozp_unit ${t})
  endif()
endforeach()

if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tests/oom/oom_test.c)
  add_executable(ozp_oom_test tests/oom/oom_test.c)
  target_link_libraries(ozp_oom_test PRIVATE ozenproto_lib)
endif()

if(OZP_FUZZ)
  add_executable(fuzz_expand fuzz/fuzz_expand.c)
  target_compile_options(fuzz_expand PRIVATE -fsanitize=fuzzer)
  target_link_options(fuzz_expand PRIVATE -fsanitize=fuzzer)
  target_link_libraries(fuzz_expand PRIVATE ozenproto_lib)
endif()
